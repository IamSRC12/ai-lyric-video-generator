    setActiveNav(index);
    const refs = [keySectionRef, audioSectionRef, lyricsSectionRef, clipsSectionRef];

  };

  async function validateKey() {
    setKeyError(null); setValidating(true);
    try {
      const activeKey = key.trim();
      if (!activeKey) { setKeyError("Please enter a Groq API key."); setKeyOk(false); setValidating(false); return; }

      const json = (await res.json()) as { ok?: boolean; error?: string };
      if (!res.ok || !json.ok) { setKeyError(json.error ?? "Could not validate that key."); setKeyOk(false); return; }
