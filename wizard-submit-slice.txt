  async function analyze() {
    if (!audioFile) { await generateWithoutAudio(); return; }
    setRunning(true);
    setStartedAt(Date.now());
    setEtaMs(Math.round((6 + duration * 0.22) * 1000)); // initial estimate
    setLog(["Starting speech & DSP alignment pipeline..."]);
    setProgress(0.1);
    const form = new FormData();
    form.set("lyrics", lyrics); form.set("durationSec", String(duration));
    form.set("title", songTitle.trim() || parsed.title || audioFile.name.replace(/\.[^.]+$/, ""));
    form.set("artist", artistName.trim() || parsed.artist || "Unknown Artist"); form.set("language", "en");

    try {
      let masterMono = masterMonoRef.current;
      if (!masterMono) {
        const decoded = masterAudioBufferRef.current || (await decodeAudio(await audioFile.arrayBuffer()));
        masterAudioBufferRef.current = decoded;
        masterMono = downsampleMono(decoded, 16000);
        masterMonoRef.current = masterMono;
      }
      const wav = encodeWavMono16(masterMono, 16000);
      form.set("audio", new File([wav], "master.wav", { type: "audio/wav" }));

      // Fast collect anchors (reuse already locked clips, or correlate remaining)
      const anchors: AnchorHit[] = [];
      for (const clip of clips) {
        if (clip.lineIndex < 0 && !clip.isInstrumental) continue;
        if (clip.fftLocked && clip.offsetSec !== undefined) {
          anchors.push({
            lineIndex: clip.lineIndex,
            offsetSec: clip.offsetSec,
            durationSec: clip.durationSec,
            peakConfidence: clip.matchScore,
          });
        } else {
          try {
            const clipDecoded = await decodeAudio(await clip.file.arrayBuffer());
            const clipMono = downsampleMono(clipDecoded, 16000);
            const hit = crossCorrelate(masterMono, clipMono, 16000);
            if (hit.peakConfidence >= 0.35) {
              anchors.push({
                lineIndex: clip.lineIndex,
                offsetSec: hit.offsetSec,
                durationSec: clipDecoded.duration,
                peakConfidence: hit.peakConfidence,
              });
            }
          } catch {}
        }
      }

      if (anchors.length > 0) {
        setLog((curr) => [...curr, `✓ Locked ${anchors.length} cut audio clips with FFT precision!`]);
        form.set("anchors", JSON.stringify(anchors));
      }

      const res = await fetch("/api/analyze", { method: "POST", body: form });
      if (!res.body) { setLog(["The analyze stream failed."]); setRunning(false); return; }

      const reader = res.body.getReader();
      const decoder = new TextDecoder();
      let buf = ""; let captions: Caption[] = []; let flagged = 0;
      let title = songTitle.trim() || parsed.title || "Untitled";
      let artist = artistName.trim() || parsed.artist || "Unknown Artist";
      let streamFailed = false;

      while (true) {
